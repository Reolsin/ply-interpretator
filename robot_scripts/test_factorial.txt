res = 0 FUNCTION sum(first = 0, second = 0) {
    WHILE second GT 0 DO {
        INC first

}

    INC s
    WHILE (DEC s) GT 0 DO {
        [res] = sum(res, f)
    }
}

res = 0 FUNCTION factorial(int = 1) {
    UINT tmp = int
    IF int GT 1 {
        [tmp] = factorial(DEC tmp)
        [res] = mul(int, tmp)
    } ELSE {
        res = 1
    }
}

    UINT i = 1
